#!/bin/bash
export sum=0
export cnt=0
find -type f | ( while read f; do
    if file "$f" | grep -q "ASCII text"; then
        c=`cat "$f"|wc -l`
        sum=$(($sum+$c))
        cnt=$(($cnt+1))
    fi
done
echo "sum: $sum; cnt: $cnt; avg: $(($sum/$cnt))" )

#!/bin/bash

ARGS=$@

while true; do
	read FILE
	WORDS=""
	for WORD in $ARGS; do
		if ! grep -E -q "\<$WORD\>" $FILE; then
			WORDS+="$WORD "
		fi
	done
	ARGS=$WORDS
	if test ! -n "$ARGS"; then
		exit
	fi
done














#!/bin/bash
#Scrie un program simplu în Shell care primește oricâte directoare ca parametri, și afișează pentru fiecare dintre ele:
#Director XXXXXX
#fișiere executabile: număr
#Fișiere neexecutabile: număr
#Directoare : număr
#Se vor efectua toate validările necesare

if [ $# -eq 0 ]; then
        echo "nu ai argumente"
        exit
fi

count=0
while [ $# -ne 0 ]; do
        dir=$1
        if [ ! -d $dir ]; then
                echo "nu e director"
                shift 1
                continue
        else
                echo "Directory: $dir"
                count=$((count + 1))
                echo "Numar directoare $count"
                # Număr fișiere executabile (excludem directoarele)
                exec_files=$(find "$dir" -maxdepth 1 -type f -executable | wc -l)

                # Număr fișiere neexecutabile (doar fișiere)
                non_exec_files=$(find "$dir" -maxdepth 1 -type f ! -executable | wc -l)

                # Număr subdirectoare (excludem directorul în sine)
                subdirs=$(find "$dir" -mindepth 1 -maxdepth 1 -type d | wc -l)
                echo "Executabile: $exec_files"
                echo "Neexecutabile: $non_exec_files"
                echo "subdirs: $subdirs"
        fi
        shift 1
done
