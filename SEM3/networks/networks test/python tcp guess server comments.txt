import socket
import select
import time
import random

# ---------------- CONFIGURATION ----------------
HOST = "0.0.0.0"
PORT = 5000
MIN_CLIENTS = 1
STATE_BROADCAST_INTERVAL = 1.0
# ------------------------------------------------


def broadcast(clients, message):
    """Send a message to all connected clients."""
    for c in clients:
        try:
            c.sendall(message.encode())
        except:
            pass


def main():
    server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    server.bind((HOST, PORT))
    server.listen()

    print(f"Server listening on {HOST}:{PORT}")

    clients = []
    current_turn = 0
    target_number = None
    last_broadcast = 0
    game_active = False

    while True:
        readable, _, exceptional = select.select([server] + clients, [], clients, 0.5)

        # Handle new connections
        for s in readable:
            if s is server:
                conn, addr = server.accept()
                clients.append(conn)
                print(f"Client connected from {addr}")
                conn.sendall(f"Welcome, you are Client {len(clients)}!\n".encode())

                if game_active:
                    conn.sendall(b"Game already in progress. Please wait for your turn.\n")
                else:
                    conn.sendall(b"Waiting for enough players to start...\n")
                continue

            # Handle data from clients
            try:
                data = s.recv(1024)
                if not data:
                    print("Client disconnected.")
                    if s in clients:
                        idx = clients.index(s)
                        clients.remove(s)
                        print(f"Removed Client {idx+1}")
                    s.close()
                    continue

                if not game_active:
                    s.sendall(b"Game not started yet.\n")
                    continue

                # Only current client can make a guess
                if clients[current_turn] != s:
                    s.sendall(b"Not your turn!\n")
                    continue

                guess_text = data.decode().strip()
                if not guess_text.isdigit():
                    s.sendall(b"Please send a valid number.\n")
                    continue

                guess = int(guess_text)
                print(f"Client {current_turn+1} guessed {guess}")

                if guess == target_number:
                    broadcast(clients, f"🎉 Client {current_turn+1} guessed correctly! The number was {target_number}.\n")
                    print("Game over. Restarting...")
                    target_number = random.randint(1, 1000)
                    broadcast(clients, f"\n🎯 New game started! Guess the new number (1–1000).\n")
                    print(f"New target number is {target_number}")
                    current_turn = 0
                    broadcast(clients, f"It is now Client 1's turn.\n")
                    clients[0].sendall(b"Your turn!\n")
                    continue
                elif guess < target_number:
                    s.sendall(b"Too low!\n")
                else:
                    s.sendall(b"Too high!\n")

                # Switch to next client
                current_turn = (current_turn + 1) % len(clients)
                broadcast(clients, f"It is now Client {current_turn+1}'s turn.\n")
                clients[current_turn].sendall(b"Your turn!\n")

            except Exception as e:
                print("Client error:", e)
                if s in clients:
                    clients.remove(s)
                s.close()

        # Start the game when enough clients are connected
        now = time.time()
        if now - last_broadcast > STATE_BROADCAST_INTERVAL:
            last_broadcast = now

            if not game_active and len(clients) >= MIN_CLIENTS:
                game_active = True
                target_number = random.randint(1, 1000)
                print(f"Starting game. Target number is {target_number}")
                broadcast(clients, "\n🎮 Game started! Guess a number between 1 and 1000.\n")
                broadcast(clients, f"It is now Client 1's turn.\n")
                clients[0].sendall(b"Your turn!\n")

        for s in exceptional:
            if s in clients:
                clients.remove(s)
            s.close()


if __name__ == "__main__":
    main()
