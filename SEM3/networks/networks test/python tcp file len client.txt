import socket

s = socket.socket()
s.connect(('127.0.0.1', 1276))

# Read file content
filename = input("Enter the filename to send: ")
with open(filename, 'r') as f:
    file_content = f.read()

# Send file length and file content
s.send(len(file_content).to_bytes(4, 'big'))
s.send(file_content.encode())

# Receive file length back from server
length_bytes = s.recv(4)
received_length = int.from_bytes(length_bytes, 'big')
print(f"Server reports file length: {received_length} bytes")

s.close()
