import socket
import select
import random
import time

SERVER_IP = "127.0.0.1"
SERVER_PORT = 7003
ACTION_INTERVAL = 1.5  # seconds between actions

def main():
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.connect((SERVER_IP, SERVER_PORT))
    sock.setblocking(False)
    print("🤖 Trader bot connected to market.")

    next_action = time.time() + ACTION_INTERVAL
    state = "WAITING"

    stocks = {}
    while True:
        readable, _, _ = select.select([sock], [], [], 0.5)

        for s in readable:
            data = s.recv(1024)
            if not data:
                print("Server closed connection.")
                return

            msg = data.decode().strip()
            print("📩", msg)

            if msg.startswith("UPDATE:"):
                try:
                    _, rest = msg.split(":", 1)
                    parts = rest.split(":")[1]
                    pairs = parts.split("|")
                    stocks = {p.split(":")[0]: float(p.split(":")[1]) for p in pairs}
                except Exception as e:
                    print("Parse error:", e)

            elif msg.startswith("STATE:END"):
                print("🏁 Market ended. Disconnecting.")
                sock.close()
                return

        now = time.time()
        if now >= next_action and stocks:
            next_action = now + ACTION_INTERVAL
            stock = random.choice(list(stocks.keys()))
            action = random.choice(["BUY", "SELL"])
            amount = random.randint(1, 3)
            sock.sendall(f"BID:{action}:{stock}:{amount}\n".encode())
            print(f"📤 Sent bid: {action} {amount}x{stock}")

if __name__ == "__main__":
    main()
