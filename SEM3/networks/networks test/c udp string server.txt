#include <sys/types.h>
#include <stdio.h>
#include <ctype.h>
#include <sys/socket.h>
#include <stdlib.h>
#include <netinet/in.h>
#include <string.h>
#include <arpa/inet.h>
#include <unistd.h>
#include <sys/wait.h>

int
validate_number (char *str)
{
  while (*str)
    {
      if (!isdigit (*str))
	{			//if the character is not a number, return
	  return 0;
	}
      str++;			//point to next character
    }
  return 1;
}


int
main (int argc, char **argv)
{
  //Validating cmd arguments

  if (argc < 2)
    {
      printf ("Give me a port.\n");
      return 1;
    }

  if (!validate_number (argv[1]))
    {
      printf ("Port must be integer.\n");
      return 2;
    }

  //Initializing socket

  int server_fd;

  struct sockaddr_in server_addr, client_addr;

  memset(&client_addr, 0, sizeof(struct sockaddr_in));

  printf ("Creating socket.\n");

  if ((server_fd = socket (AF_INET, SOCK_DGRAM, 0)) == 0)
    {
      perror ("Socket creation failed\n");
      exit (EXIT_FAILURE);
    }

  server_addr.sin_family = AF_INET;
  server_addr.sin_addr.s_addr = INADDR_ANY;
  //Seting socket to listen on all available ip addresses
  //If you want the server to listen on a single ip address use the line below insead
  //inet_aton("1.2.3.4", &server_addr.sin_addr)

  //Setting bind port
  server_addr.sin_port = htons( atoi (argv[1]));

  printf ("Binding socket to port %s\n", argv[1]);

  //Binding server

  if (bind (server_fd, (struct sockaddr *) &server_addr, sizeof (server_addr))
      < 0)
    {
      perror ("Bind error.\n");
      exit (EXIT_FAILURE);
    }

  printf("Waiting for clients...");
  
  unsigned int len = sizeof(client_addr);
  
  while (1)
    {
      int size, recv;
      
      recv = recvfrom(server_fd, &size, sizeof(int), MSG_WAITALL, ( struct sockaddr *) &client_addr, &len); 
    
      size = ntohl (size);
    
      printf("Received from %s : %d\n", inet_ntoa (client_addr.sin_addr), size);
      
      char *string;
      
      string = malloc(size * sizeof(char));
      
      recv = recvfrom(server_fd, string, size * sizeof(char), MSG_WAITALL, ( struct sockaddr *) &client_addr, &len); 

      printf("Received from %s :\n %s\n", inet_ntoa (client_addr.sin_addr), string);
      
      free(string);
    
    }

  return 0;
}
