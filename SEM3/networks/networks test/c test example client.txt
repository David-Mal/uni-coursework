import socket
import threading
import random
import struct
import time

HOST = '127.0.0.1'
PORT = 5000

even_number = random.choice([2, 4, 6, 8])
odd_number = random.choice([1, 3, 5, 7, 9])
current_state = "standby"

def listen_server(s):
    global current_state
    while True:
        try:
            data = s.recv(10)  # Receive state updates or numbers
            if not data:
                break
            try:
                # Try to decode state
                state = data.decode()
                if state in ["even", "odd", "standby", "finished"]:
                    current_state = state
                    print(f"Server state changed: {current_state}")
            except:
                # If not state, it is a number
                num = struct.unpack('B', data)[0]
                print(f"Server collected your number: {num}")
        except:
            break

if __name__ == "__main__":
    s = socket.socket()
    s.connect((HOST, PORT))

    # Send client numbers first (even, odd)
    s.sendall(struct.pack('BB', even_number, odd_number))
    print(f"My numbers - even: {even_number}, odd: {odd_number}")

    threading.Thread(target=listen_server, args=(s,), daemon=True).start()

    while current_state != "finished":
        if current_state in ["even", "odd"]:
            # Send the requested number every 2 seconds
            if current_state == "even":
                s.sendall(struct.pack('B', even_number))
            else:
                s.sendall(struct.pack('B', odd_number))
        time.sleep(2)

    s.close()
    print("Client finished.")

import socket
import threading
import random
import struct
import time

HOST = '127.0.0.1'
PORT = 5000

even_number = random.choice([2, 4, 6, 8])
odd_number = random.choice([1, 3, 5, 7, 9])
current_state = "standby"

sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
server_addr = (HOST, PORT)

# Send initial registration (even, odd) to server
sock.sendto(struct.pack('BB', even_number, odd_number), server_addr)
print(f"My numbers - even: {even_number}, odd: {odd_number}")

def listen_server():
    global current_state
    while True:
        try:
            data, addr = sock.recvfrom(1024)
            if not data:
                break
            try:
                # Try to decode server state
                state = data.decode()
                if state in ["even", "odd", "standby", "finished"]:
                    current_state = state
                    print(f"Server state changed: {current_state}")
                    if state == "finished":
                        break
            except:
                # If not state, it's acknowledgment of number
                num = struct.unpack('B', data)[0]
                print(f"Server collected your number: {num}")
        except:
            continue

# Start listening to server in a separate thread
threading.Thread(target=listen_server, daemon=True).start()

# Main loop: send requested number every 2 seconds
while current_state != "finished":
    if current_state in ["even", "odd"]:
        num = even_number if current_state == "even" else odd_number
        sock.sendto(struct.pack('B', num), server_addr)
    time.sleep(2)

sock.close()
print("Client finished.")
