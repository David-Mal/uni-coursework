import socket
import os

HOST = '127.0.0.1'
PORT = 7000

file_path = input("Enter the full path of the file: ")

with socket.socket() as s:
    s.connect((HOST, PORT))

    # Send length of file path first
    s.send(len(file_path).to_bytes(4, 'big'))
    s.send(file_path.encode())

    # Receive file length
    length_bytes = s.recv(4)
    length = int.from_bytes(length_bytes, 'big', signed=True)

    if length == -1:
        print("File does not exist on server.")
    else:
        # Receive the content
        content = b''
        while len(content) < length:
            data = s.recv(length - len(content))
            if not data:
                break
            content += data

        # Save to a new file with -copy suffix
        base_name = os.path.basename(file_path)
        new_file_name = f"{base_name}-copy"
        with open(new_file_name, 'wb') as f:
            f.write(content)
        print(f"File received and saved as {new_file_name}")
