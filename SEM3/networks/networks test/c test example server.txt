import socket
import select
import struct

HOST = '127.0.0.1'
PORT = 5000
MIN_CLIENTS = 3
MAX_TOTAL = 9000
STATE_SWITCH_COUNT = 100

server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
server_socket.bind((HOST, PORT))
server_socket.listen(10)
server_socket.setblocking(False)

sockets_list = [server_socket]
clients = {}  # socket: (even_num, odd_num)

internal_value = 0
data_collected_count = 0
switch_counter = 0
current_state = "standby"

def broadcast_state(state):
    for sock in clients.keys():
        try:
            sock.sendall(state.encode())
        except:
            pass

print(f"TCP server listening on {HOST}:{PORT}")

while internal_value <= MAX_TOTAL:
    read_sockets, _, exception_sockets = select.select(sockets_list, [], sockets_list, 0.1)

    for notified_socket in read_sockets:
        if notified_socket == server_socket:
            # New connection
            client_socket, client_address = server_socket.accept()
            client_socket.setblocking(False)
            sockets_list.append(client_socket)
        else:
            try:
                data = notified_socket.recv(1024)
                if not data:
                    sockets_list.remove(notified_socket)
                    if notified_socket in clients:
                        del clients[notified_socket]
                    notified_socket.close()
                    continue

                # If client not registered yet, first 2 bytes are even/odd numbers
                if notified_socket not in clients:
                    if len(data) >= 2:
                        even_num, odd_num = struct.unpack('BB', data[:2])
                        clients[notified_socket] = (even_num, odd_num)
                        print(f"Registered client {notified_socket.getpeername()} even={even_num}, odd={odd_num}")
                        data = data[2:]  # remove registration bytes

                # Only start collecting if minimum clients connected
                if len(clients) < MIN_CLIENTS:
                    continue

                # Collect data from client
                for byte in data:
                    num = byte
                    internal_value += num
                    data_collected_count += 1
                    switch_counter += 1
                    print(f"Received {num}, internal_value={internal_value}")

                    if switch_counter >= STATE_SWITCH_COUNT:
                        current_state = "odd" if current_state == "even" else "even"
                        broadcast_state(current_state)
                        switch_counter = 0

            except:
                sockets_list.remove(notified_socket)
                if notified_socket in clients:
                    del clients[notified_socket]
                notified_socket.close()

    for notified_socket in exception_sockets:
        sockets_list.remove(notified_socket)
        if notified_socket in clients:
            del clients[notified_socket]
        notified_socket.close()

    # Start collection when minimum clients connected
    if current_state == "standby" and len(clients) >= MIN_CLIENTS:
        current_state = "even"
        broadcast_state(current_state)

print("Data collection finished!")
broadcast_state("finished")
for sock in clients.keys():
    sock.close()
server_socket.close()

import socket
import select
import struct

HOST = '127.0.0.1'
PORT = 5000
MIN_CLIENTS = 3
MAX_TOTAL = 9000
STATE_SWITCH_COUNT = 100

udp_socket = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
udp_socket.bind((HOST, PORT))
udp_socket.setblocking(False)

clients = {}  # addr: (even_num, odd_num)

internal_value = 0
data_collected_count = 0
switch_counter = 0
current_state = "standby"

def broadcast_state(state):
    for addr in clients.keys():
        try:
            udp_socket.sendto(state.encode(), addr)
        except:
            pass

print(f"UDP server listening on {HOST}:{PORT}")

while internal_value <= MAX_TOTAL:
    read_sockets, _, _ = select.select([udp_socket], [], [], 0.1)

    for sock in read_sockets:
        data, addr = sock.recvfrom(1024)

        if addr not in clients:
            if len(data) >= 2:
                even_num, odd_num = struct.unpack('BB', data[:2])
                clients[addr] = (even_num, odd_num)
                print(f"Registered client {addr} even={even_num}, odd={odd_num}")
            continue

        if len(clients) < MIN_CLIENTS:
            continue

        even_num, odd_num = clients[addr]
        if current_state == "even":
            num = even_num
        elif current_state == "odd":
            num = odd_num
        else:
            continue

        internal_value += num
        data_collected_count += 1
        switch_counter += 1
        print(f"Received {num} from {addr}, internal_value={internal_value}")

        # Send acknowledgment
        udp_socket.sendto(struct.pack('B', num), addr)

        if switch_counter >= STATE_SWITCH_COUNT:
            current_state = "odd" if current_state == "even" else "even"
            broadcast_state(current_state)
            switch_counter = 0

    # Start collection when minimum clients connected
    if current_state == "standby" and len(clients) >= MIN_CLIENTS:
        current_state = "even"
        broadcast_state(current_state)

print("Data collection finished!")
broadcast_state("finished")
udp_socket.close()
