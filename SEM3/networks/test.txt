(defun fib(n)
"print nth fibo number"
(if (< n 2)
  n
  (+ (fib(- n 1))
     (fib(- n 2))
  )
))

;; Write a function to return the n-th element of a list, or NIL if such an element does not exist
(defun nth-elem(x n)
  (cond ((null x)
      NIL)
      ((= n 1) (car x))
  (t (nth-elem (cdr x) (- n 1)))))
  
(print (nth-elem '(1 2 3 4 5) 3))
