Table User: id(int),username(string),nativeLanguage(string)
table Exercise: id(int),prompt(string),correctAnswer(string),difficulty(int),topic(string) (difficulty is 1 2 3 (as in easy medium hard)
table Result: id(int),userId(int),exerciseId(int),givenAnswer(string),correct(int),atteptedAt(datetime)
The user should authentificate prior to using the applicaion by specifying their username (we assume the username wxists in the USer table,no checks are required)

client side for the duration of the working session.
The user can start submiting answers to exercises.after each submitted answer the client  or server updates the score ,ap locally:

USE adaptive_language
CREATE TABLE User (
    id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(100),
    nativeLanguage(100)
);

CREATE TABLE Exercise (
    id INT AUTO_INCREMENT PRIMARY KEY,
    prompt VARCHAR(255),
    correctAnswer VARCHAR(100),
    difficulty INT,
    topic VARCHAR(100)
);

CREATE TABLE Result (
    id INT AUTO_INCREMENT PRIMARY KEY,
    userId INT,
    exerciseId INT,
    givenAnswer VARCHAR(100),
    correct INT,
    attemptedAt DATETIME
);
